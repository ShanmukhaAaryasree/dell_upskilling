STEP 8: CTE - TOTAL SPEND PER CUSTOMER

scala> spark.sql("""
     | WITH customer_spend AS (
     |   SELECT customer_id, SUM(amount) AS total_spend
     |   FROM orders
     |   WHERE status = 'Completed'
     |   GROUP BY customer_id
     | )
     | SELECT c.name, c.city, cs.total_spend
     | FROM customers c
     | INNER JOIN customer_spend cs
     |   ON c.customer_id = cs.customer_id

STEP 9: AGGREGATION USING HAVING

scala> spark.sql("""
     | SELECT c.city, SUM(o.amount) AS city_total
     | FROM customers c
     | INNER JOIN orders o

     | WHERE o.status = 'Completed'
     | GROUP BY c.city

STEP 14: ROW_NUMBER, RANK AND DENSE_RANK

scala> spark.sql("""
     | WITH customer_spend AS (
     |   SELECT c.customer_id, c.name, c.city,
     |          COALESCE(SUM(o.amount), 0) AS total_spend
     |   FROM customers c
     |   LEFT JOIN orders o
     |     ON c.customer_id = o.customer_id
     |    AND o.status = 'Completed'
     |   GROUP BY c.customer_id, c.name, c.city
     | )
     | SELECT name, city, total_spend,
     |        ROW_NUMBER() OVER (
     |          PARTITION BY city ORDER BY total_spend DESC
     |        ) AS rn,
     |        RANK() OVER (
     |          PARTITION BY city ORDER BY total_spend DESC
     |        ) AS rnk,
     |        DENSE_RANK() OVER (
     |          PARTITION BY city ORDER BY total_spend DESC
     |        ) AS drnk
     | FROM customer_spend
     | ORDER BY city, total_spend DESC
     | """).show()

+-------------+--------+-----------+---+---+----+
|         name|    city|total_spend| rn|rnk|drnk|
+-------------+--------+-----------+---+---+----+
| Eva Martinez| Chicago|      200.0|  1|  1|   1|
|    Bob Smith| Chicago|        0.0|  2|  2|   2|
|    David Kim| Boston |        0.0|  1|  1|   1|

STEP 20: JSON PARSING AND EXPLODE

     | WITH parsed_items AS (

     |          FROM_JSON(items_json,
     |            'ARRAY<STRUCT<product: STRING, qty: INT, price: DOUBLE>>') AS items
     |   FROM order_items_raw

     |   SELECT p.order_id, item.product, item.qty, item.price

STEP 22: CHALLENGE - TOP SPENDER PER CITY

     | WITH customer_spend AS (
     |   SELECT c.customer_id, c.city, c.name,
     |          COALESCE(SUM(o.amount), 0) AS total_spend
     |   FROM customers c
     |   LEFT JOIN orders o
     |     ON c.customer_id = o.customer_id

     |   SELECT city, name, total_spend,

STEP 24: CHALLENGE - RUNNING TOTAL PER CUSTOMER

scala> spark.sql("""
     | WITH parsed_items AS (
     |   SELECT order_id,
     |          FROM_JSON(items_json,
     |            'ARRAY<STRUCT<product: STRING, qty: INT, price: DOUBLE>>') AS items
     |   FROM order_items_raw
     | ),
     | exploded_items AS (
     |   SELECT p.order_id, item.product, item.qty, item.price
     |   FROM parsed_items p
     |   LATERAL VIEW EXPLODE(p.items) e AS item
     | ),
     | order_lines AS (
     |   SELECT c.customer_id, c.name, c.city,
     |          o.order_id, o.order_date,
     |          i.product, i.qty, i.price,
     |          i.qty * i.price AS line_total
     |   FROM exploded_items i
     |   INNER JOIN orders o ON i.order_id = o.order_id
     |   INNER JOIN customers c ON o.customer_id = c.customer_id
     | )
     | SELECT name, city, order_id, product, qty, price, line_total,
     |        SUM(line_total) OVER (
     |          PARTITION BY customer_id
     |          ORDER BY order_date, order_id, product
     |          ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
     |        ) AS running_total
     | FROM order_lines
     | ORDER BY name, order_date, order_id, product
     | """).show()

|         name|    city|order_id| product|qty|price|line_total|running_total|
