scala> spark.sql("""
     | WITH company_avg AS (
     |   SELECT AVG(salary) AS avg_sal
     |   FROM employees
     | ),
     | above_avg AS (
     |   SELECT e.emp_id, e.name, e.salary, c.avg_sal
     |   FROM employees e
     |   CROSS JOIN company_avg c

STEP 29: EXERCISE 9A – SQL VS DATAFRAME API
SOLUTION A: PURE SPARK SQL


scala> spark.sql("""
     | WITH ranked AS (
     |   SELECT e.name, e.salary, d.dept_name,
     |          ROW_NUMBER() OVER (
     |            PARTITION BY e.dept_id
     |            ORDER BY e.salary DESC
     |          ) AS rn
     |   FROM employees e
     |   JOIN departments d ON e.dept_id = d.dept_id
     | )
     | SELECT dept_name, name AS top_employee, salary
     | FROM ranked
     | WHERE rn = 1
