package com.training

import org.apache.spark.sql.SparkSession

object ReadParquet {
  def main(args: Array[String]): Unit = {

    val spark = SparkSession.builder()
      .appName("ReadParquet")
      .master("local[*]")
      .getOrCreate()

    val df = spark.read
      .parquet("C:/Users/Itinfra/IdeaProjects/ClickstreamWindowedJob/output/parquet/clickstream_windows")

    df.show(false)

    spark.stop()
  }
}
