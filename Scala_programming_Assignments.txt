scala> def validateCustomer(customer: Customer): Either[String, Customer] = {
     |   customer match {
     |     case Customer(Some(id), Some(name), Some(age), Some(email))
     |       if id > 0 && name.trim.nonEmpty && age >= 18 && age <= 100 && email.contains("@") =>
     |         Right(customer)
     |     case Customer(None, _, _, _) => Left("Customer ID is missing")
     |     case Customer(Some(id), _, _, _) if id <= 0 => Left("Customer ID must be positive")
     |     case Customer(_, None, _, _) => Left("Name is missing")
     |     case Customer(_, Some(name), _, _) if name.trim.isEmpty => Left("Name must not be empty")
     |     case Customer(_, _, None, _) => Left("Age is missing")
     |     case Customer(_, _, Some(age), _) if age < 18 || age > 100 => Left("Age must be between 18 and 100")
     |     case Customer(_, _, _, None) => Left("Email is missing")
     |     case Customer(_, _, _, Some(email)) if !email.contains("@") => Left("Email must contain @")
     |   }
     | }
def validateCustomer(customer: Customer): Either[String,Customer]
